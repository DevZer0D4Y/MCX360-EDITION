# Host-side tests for the launcher: its checks on the player's files, the
# importer, and the disc image and title update package readers. Builds on
# macOS or Linux, independently of the iOS app:
#
#   cmake -S app/tests -B app/out/tests && cmake --build app/out/tests && ctest --test-dir app/out/tests
#
# With -DREXGLUE_SDK=/path/to/patched/rexglue-sdk (needs its fmt, spdlog,
# toml++, xxHash, utfcpp, simde and CLI11 submodules) it also builds
# sdk_reader_test, which runs the importer with the SDK's real STFS reader and
# compares the app's disc image reader with the SDK's.
#
# The test data is synthetic, generated by tools/*.py; no game files needed.

cmake_minimum_required(VERSION 3.25)
project(mc360_tests LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
enable_testing()

set(APP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(TOOLS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tools")
set(REXGLUE_SDK "$ENV{REXGLUE_SDK}" CACHE PATH "Patched rexglue-sdk checkout (optional)")

option(MC360_TEST_SANITIZE "Build the tests with AddressSanitizer and UBSan" ON)
if(MC360_TEST_SANITIZE AND NOT MSVC)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

set(LAUNCHER_SOURCES
    ${APP_DIR}/src/launcher/importer.cpp
    ${APP_DIR}/src/launcher/launcher.cpp
    ${APP_DIR}/src/launcher/xdvdfs.cpp
    ${APP_DIR}/src/launcher/xex_info.cpp
)

add_executable(launcher_test launcher_test.cpp
    ${APP_DIR}/src/launcher/launcher.cpp
    ${APP_DIR}/src/launcher/xex_info.cpp)
target_include_directories(launcher_test PRIVATE ${APP_DIR}/src)
add_test(NAME launcher COMMAND launcher_test ${TOOLS_DIR}/mkxex.py)

add_executable(importer_test importer_test.cpp ${LAUNCHER_SOURCES})
target_include_directories(importer_test PRIVATE ${APP_DIR}/src)
target_link_libraries(importer_test PRIVATE Threads::Threads)
add_test(NAME importer COMMAND importer_test ${TOOLS_DIR})

if(REXGLUE_SDK)
    # Just the slice of the SDK the readers need, compiled into the test.
    set(SDK_SOURCES
        ${REXGLUE_SDK}/src/core/cvar.cpp
        ${REXGLUE_SDK}/src/core/filesystem.cpp
        ${REXGLUE_SDK}/src/core/filesystem_wildcard.cpp
        ${REXGLUE_SDK}/src/core/logging.cpp
        ${REXGLUE_SDK}/src/core/mutex.cpp
        ${REXGLUE_SDK}/src/core/string_buffer.cpp
        ${REXGLUE_SDK}/src/core/utf8.cpp
        ${REXGLUE_SDK}/src/core/filesystem_posix.cpp
        ${REXGLUE_SDK}/src/core/mapped_memory_posix.cpp
        ${REXGLUE_SDK}/src/core/platform/env_posix.cpp
        ${REXGLUE_SDK}/src/filesystem/device.cpp
        ${REXGLUE_SDK}/src/filesystem/entry.cpp
        ${REXGLUE_SDK}/src/filesystem/devices/disc_image_device.cpp
        ${REXGLUE_SDK}/src/filesystem/devices/disc_image_entry.cpp
        ${REXGLUE_SDK}/src/filesystem/devices/disc_image_file.cpp
        ${REXGLUE_SDK}/src/filesystem/devices/stfs_container_device.cpp
        ${REXGLUE_SDK}/src/filesystem/devices/stfs_container_entry.cpp
        ${REXGLUE_SDK}/src/filesystem/devices/stfs_container_file.cpp
    )
    add_executable(sdk_reader_test sdk_reader_test.cpp ${LAUNCHER_SOURCES}
        ${APP_DIR}/src/launcher/stfs_archive.cpp ${SDK_SOURCES})
    target_include_directories(sdk_reader_test PRIVATE
        ${APP_DIR}/src
        ${REXGLUE_SDK}/include
        ${REXGLUE_SDK}/src/core
        ${REXGLUE_SDK}/src/filesystem/devices
        ${REXGLUE_SDK}/thirdparty/cli11/include
        ${REXGLUE_SDK}/thirdparty/fmt/include
        ${REXGLUE_SDK}/thirdparty/simde
        ${REXGLUE_SDK}/thirdparty/spdlog/include
        ${REXGLUE_SDK}/thirdparty/tomlplusplus/include
        ${REXGLUE_SDK}/thirdparty/utfcpp/source
        ${REXGLUE_SDK}/thirdparty/xxHash)
    target_compile_definitions(sdk_reader_test PRIVATE SPDLOG_FMT_EXTERNAL FMT_HEADER_ONLY)
    target_link_libraries(sdk_reader_test PRIVATE Threads::Threads)
    add_test(NAME sdk_readers COMMAND sdk_reader_test ${TOOLS_DIR})

    # Core loading, with two stand-in core libraries next to the test.
    foreach(_core tu9 tu14)
        add_library(mc360core_${_core} SHARED
            ${APP_DIR}/src/core/core_entry.cpp fake_core/fake_core.cpp)
        target_include_directories(mc360core_${_core} PRIVATE
            ${APP_DIR}/src ${CMAKE_CURRENT_SOURCE_DIR}/fake_core ${REXGLUE_SDK}/include)
        target_compile_definitions(mc360core_${_core} PRIVATE MC360_CORE_ID="${_core}")
        set_target_properties(mc360core_${_core} PROPERTIES
            LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
    endforeach()
    add_executable(core_loader_test core_loader_test.cpp
        ${APP_DIR}/src/launcher/core_loader.cpp
        ${REXGLUE_SDK}/src/core/dynlib_posix.cpp
        ${SDK_SOURCES})
    target_include_directories(core_loader_test PRIVATE
        $<TARGET_PROPERTY:sdk_reader_test,INCLUDE_DIRECTORIES>)
    target_compile_definitions(core_loader_test PRIVATE SPDLOG_FMT_EXTERNAL FMT_HEADER_ONLY
        CORE_SUFFIX="${CMAKE_SHARED_LIBRARY_SUFFIX}")
    target_link_libraries(core_loader_test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    set_target_properties(core_loader_test PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
    add_dependencies(core_loader_test mc360core_tu9 mc360core_tu14)
    add_test(NAME core_loader COMMAND core_loader_test)
else()
    message(STATUS "REXGLUE_SDK not set: skipping sdk_reader_test")
endif()
